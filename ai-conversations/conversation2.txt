def get_weather_data(location, forecast_days=5):
    """
    Retrieve weather data for a specified location.
    
    Args:
        location (str): City or location name
        forecast_days (int): Number of days to forecast (1-5)
        
    Returns:
        dict: Weather data including current conditions and forecast
              Returns None if there's an error
    """
    
    # SECTION 1: Input Validation
    # Ensures forecast_days is within valid range to prevent API errors
    if not isinstance(forecast_days, int) or forecast_days < 1 or forecast_days > 5:
        print(f"Warning: forecast_days must be between 1 and 5. Using default value of 5.")
        forecast_days = 5
    
    try:
        # SECTION 2: API Call
        # The get_weather function from fetch_my_weather handles the actual API request
        weather_result = get_weather(location, forecast_days=forecast_days)
        
        # SECTION 3: Check for null response
        # API might return None for invalid locations or connection issues
        if weather_result is None:
            print(f"Error: Could not retrieve weather data for '{location}'")
            return None
        
        # SECTION 4: Structure current conditions
        # Extract nested data safely using .get() to avoid KeyErrors
        structured_data = {
            'location': location,
            'current': {
                'temperature': weather_result.get('current_condition', [{}])[0].get('temp_C', 'N/A'),
                'feels_like': weather_result.get('current_condition', [{}])[0].get('FeelsLikeC', 'N/A'),
                'description': weather_result.get('current_condition', [{}])[0].get('weatherDesc', [{}])[0].get('value', 'N/A'),
                'humidity': weather_result.get('current_condition', [{}])[0].get('humidity', 'N/A'),
                'wind_speed': weather_result.get('current_condition', [{}])[0].get('windspeedKmph', 'N/A'),
                'precipitation': weather_result.get('current_condition', [{}])[0].get('precipMM', 'N/A')
            },
            'forecast': []
        }
        
        # SECTION 5: Process forecast data
        # Loop through each day's forecast and extract relevant information
        forecast_data = weather_result.get('weather', [])
        for day in forecast_data[:forecast_days]:
            forecast_entry = {
                'date': day.get('date', 'N/A'),
                'max_temp': day.get('maxtempC', 'N/A'),
                'min_temp': day.get('mintempC', 'N/A'),
                'avg_temp': day.get('avgtempC', 'N/A'),
                'description': day.get('hourly', [{}])[0].get('weatherDesc', [{}])[0].get('value', 'N/A'),
                'precipitation_mm': day.get('hourly', [{}])[0].get('precipMM', '0'),
                'chance_of_rain': day.get('hourly', [{}])[0].get('chanceofrain', '0'),
                'humidity': day.get('hourly', [{}])[0].get('humidity', 'N/A')
            }
            structured_data['forecast'].append(forecast_entry)
        
        return structured_data
        
    except Exception as e:
        # SECTION 6: Catch-all error handling
        # Ensures the function never crashes the entire application
        print(f"Error retrieving weather data: {str(e)}")
        return None

EXPLANATION OF KEY DECISIONS:

1. **Nested .get() calls**: The API returns deeply nested JSON. Using .get() with 
   defaults prevents KeyError exceptions if any field is missing.

2. **Default values**: 'N/A' for missing string data, '0' for missing numeric data 
   like precipitation.

3. **Try-except block**: Wraps the entire operation to catch unexpected errors 
   (network issues, malformed responses, etc.)

4. **Explicit None returns**: Makes it easy for calling code to check if the 
   operation succeeded.
